### 2. 点击穿透：**轮询 + 渲染进程上报的命中区**（核心）

```js
// —— 命中矩形判定（支持矩形数组：一个宠物可以有多个可点区域）——
function isPointInsidePetBounds(pt, b) {
  return pt.x >= b.x && pt.x <= b.x + b.width && pt.y >= b.y && pt.y <= b.y + b.height
}
function isPointInsidePetHitBounds(pt, bounds) {
  if (Array.isArray(bounds)) return bounds.some(b => isPointInsidePetBounds(pt, b))
  return isPointInsidePetBounds(pt, bounds)
}
// 宠物不可见、或没上报命中区 → 一律穿透
function shouldIgnorePetMouseEvents({ petVisible, cursor, petHitBounds }) {
  if (!petVisible) return true
  if (!petHitBounds) return true
  return !isPointInsidePetHitBounds(cursor, petHitBounds)
}
```

```js
// 切换本身带**状态去重**（这是不闪的关键）
function setPetMouseEventsIgnored(ignore, opts) {
  if (!petWindow || petWindow.isDestroyed()) return
  if (petMouseIgnoreState === ignore) return          // ★★ 值没变就什么都不做
  petWindow.setIgnoreMouseEvents(ignore, ignore ? (opts ?? { forward:true }) : opts)
  petMouseIgnoreState = ignore
}

function refreshPetMousePassthrough() {
  const visible = Boolean(petWindow && !petWindow.isDestroyed() && petWindow.isVisible())
  const cursor  = screen.getCursorScreenPoint()
  const ignore  = shouldIgnorePetMouseEvents({ petVisible: visible, cursor, petHitBounds })
  setPetMouseEventsIgnored(ignore)
  // ★ 复用同一个光标源驱动视线跟随，不再另开一条定时器
  if (visible) petWindow.webContents.send('pet:cursorMoved', cursor)
}

function startPetMouseTracking() {
  if (petMouseTracker) return
  refreshPetMousePassthrough()
  petMouseTracker = setInterval(refreshPetMousePassthrough, PET_MOUSE_TRACKING_INTERVAL_MS)
}
function stopPetMouseTracking() {
  if (petMouseTracker) { clearInterval(petMouseTracker); petMouseTracker = null }
  setPetMouseEventsIgnored(true)      // 停跟踪 → 强制穿透，别挡鼠标
}
```

命中区由**渲染进程**上报（屏幕坐标，支持数组）：

```js
function updatePetHitBounds(bounds) { petHitBounds = bounds; refreshPetMousePassthrough() }  // 立即刷新
function normalizePetHitBounds(raw) {
  const list = (Array.isArray(raw) ? raw : [raw]).map(b => ({
    x: Math.round(b.x), y: Math.round(b.y),
    width: Math.max(0, Math.round(b.width)), height: Math.max(0, Math.round(b.height)),
  })).filter(b => b.width > 0 && b.height > 0)
  if (list.length === 0) return null
  return Array.isArray(raw) ? list : list[0]
}
```

窗口创建时**先设成穿透**，避免还没上报命中区就挡住桌面：

```js
petMouseIgnoreState = null
setPetMouseEventsIgnored(true, { forward: true })
```

**交互保护窗口（很聪明的一条）**：气泡/提醒刚弹出时，鼠标往往还没移进去，如果立刻判"在外→穿透"，
用户就点不到那个气泡。所以给一个 1200ms 的强制可点窗口：

```js
function guardPetInteraction(ms = 0x4b0 /* 1200 */) {
  petInteractionGuardUntil = Date.now() + ms
  refreshPetMousePassthrough()
}
// shouldIgnorePetMouseEvents 的调用方把 guard 也纳入判定
```

### 4. 光标只轮询一次，两处复用

`refreshPetMousePassthrough()` 里那次 `screen.getCursorScreenPoint()`：
① 决定穿透；② `webContents.send('pet:cursorMoved', cursor)` 驱动她的视线跟随。
**一份坐标喂两个用途**，不另开 16ms 定时器。

### 5. 拖动 = **渲染进程给增量，主进程只做 setPosition**

```js
// preload：moveBy(dx, dy, localBounds, projectedBodyCenter) → invoke('pet:moveBy', ...)
//          endDrag(localBounds)                            → invoke('pet:endDrag', ...)
function movePetBy(dx, dy, localBounds, projectedBodyCenter) {
  if (!petWindow) return
  const mx = Number.isFinite(dx) ? Math.round(dx) : 0
  const my = Number.isFinite(dy) ? Math.round(dy) : 0
  if (!mx && !my) return
  petPlacementMode = 'independent'                   // 一旦手动拖过，就不再跟随主窗口
  const win = petWindow.getBounds()
  const body = getPetBodyScreenBounds()              // = petHitBounds[0]
  const local = localBounds ?? (body
    ? { x: body.x - win.x, y: body.y - win.y, width: body.width, height: body.height }
    : { x: 0, y: 0, width: win.width, height: win.height })
  const display = getDisplayForPetDrag(screen, win, local, { dx: mx, dy: my }, projectedBodyCenter)
  const next = getBoundedPetWindowPosition(win, { dx: mx, dy: my }, display.workArea, local)
  if (next.x === win.x && next.y === win.y) return
  petWindow.setPosition(next.x, next.y)
}
```

**夹取按"她本体"而不是窗口**（和我们 `clampPet` 的思路一致，但写法更干净）：

```js
function getBoundedPetWindowPosition(win, {dx, dy}, workArea, local) {
  const nx = win.x + dx, ny = win.y + dy
  const l = local ?? { x:0, y:0, width:win.width, height:win.height }
  const minX = workArea.x - l.x
  const maxX = workArea.x + workArea.width  - l.width  - l.x
  const minY = workArea.y - l.y
  const maxY = workArea.y + workArea.height - l.height - l.y
  return { x: clamp(nx, minX, maxX), y: clamp(ny, minY, maxY) }
}
```

松开时**把本体夹进工作区并落盘**：

```js
function finishPetDrag(localBounds) {
```

### 6. 位置持久化

- 窗口 `moved` 事件 → 节流 **400ms（0x190）** 存一次；
- 松开拖动 → 立刻存一次；
- 存的是 **`{mode, ...}`，且用"她本体的屏幕位置"来表达**（`getPersistedPetWindowPosition`），
  不是裸的窗口坐标 → 换分辨率/改窗口尺寸后还能还原到"她在屏幕上的那个位置"。

### 7. Windows 透明窗口会自己坏 → 有恢复机制

`createPetWindowRecoverySupervisor` / `createPetWindowRendererRecoverySupervisor`，
配合这段"抖动重绘"：

```js
const b = petWindow.getBounds()
const tweak = { ...b, width: b.width > 1 ? b.width - 1 : b.width + 1 }   // ★ 宽度 ±1 抖一下
petWindow.hide()
try {
  petWindow.setBackgroundColor(PET_TRANSPARENT_BACKGROUND_COLOR)
  petWindow.setBounds(tweak, false)
  petWindow.setBounds(b, false)
  petWindow.webContents.invalidate?.()      // 0x319c
} finally { if (!petWindow.isDestroyed()) petWindow.show() }
```

多个延迟档位重试（`PET_WINDOW_TRANSPARENCY_RECOVERY_DELAYS_MS`），带 generation 号防止串档。
**这是 Win32 分层窗口（layered window）的经典 glitch 兜底**，不是多余的代码。

## 二、我们自己的方案（学机制，不抄实现）

### 决策总表

| 维度 | NewMax | **我们采用** | 为什么 |
| --- | --- | --- | --- |
| 窗口裁剪 | ❌ 不用区域 | ❌ **删掉 `SetWindowRgn` 成型** | 区域表达不了"菜单打开"这种动态状态；而且区域超出窗口会被系统剪掉（我们已经踩过）。它只能表达静态轮廓 |
| 命中/穿透 | 渲染进程上报命中矩形 → 主进程轮询判定 → **值变化才切** | ✅ **完全采用** | 一次解决"死区"和"闪"：死区因为真穿透而消失，闪因为去重而不再整窗重合成 |
| 命中区形状 | 矩形 / **矩形数组** | ✅ 采用，且可直接复用我们已有的掩码 `tight` | 我们已经有逐像素掩码，比它的矩形更准；数组形式正好表达"她 + 工具栏 + 打开的菜单" |
| 拖动 | 渲染进程给 `dx,dy` → 主进程 `setPosition` | ✅ **采用（替掉现在的反推+33ms）** | 反推法有反馈环、还要跟 DPR 较劲（y 轴冻住就是这么来的）；增量法天生跟手 |
| 夹取 | 按她**本体**在 workArea 内 | ✅ 采用（我们已有 `clampPet`，改成用它的写法） | 目标一致，写法更简单、不用猜窗口尺寸 |
| 光标复用 | 一次 `getCursorScreenPoint()` 喂穿透 + 视线 | ✅ 采用（**我们已有**：`main.js:427` 的 `cursorFeed` 本来就是"读全局光标喂视线"，把穿透判定并进同一个回调即可，不必另开定时器） | 省一条定时器，也避免两处读到不同帧的坐标 |
| 焦点 | `focusable:false`，需要键盘才临时开 | ✅ **采用（已核实我们现在确实缺这个）** | `main.js:178` 的 `BrowserWindow` 选项里**没有 `focusable:false`** → 默认 true → 点她会抢焦点，用户在 DSH 里打字会被打断 |
| 交互保护 | 1200ms 强制可点 | ✅ 采用 | 气泡刚弹出时鼠标还没移进去，否则点不到 |
| 透明底色 | `transparent:true` **且** `backgroundColor:'#00000000'` | ✅ 采用 | 我们只设了 `transparent:true`（`main.js:182`），没设 `backgroundColor`；而它的"抖动重绘"恢复手段正是靠 `setBackgroundColor` 才生效 |
| 透明 glitch 兜底 | 宽度 ±1 抖动 + 重设底色 + invalidate | ⚠️ **先不做，但记进坑清单** | 没实测到就先不加复杂度；一旦出现"她突然变黑/不透明"，直接照它的做法加 |
| 持久化 | 存"本体的屏幕位置" | ✅ 采用 | 我们现在存裸窗口坐标 + 启动被 `clampPet` 钉右下角 = 等于不记忆 |

### 我们的关键差异（比它更进一步）

1. **命中区用掩码，不用矩形**
   它只有矩形（贴图宠物）。我们有 `buildMask()` 的逐像素网格 + `tight`，
   可以做**逐像素穿透**：她轮廓外真的点得到桌面。这是我们的优势，别丢掉。
   实现上仍然传给主进程"一个或多个矩形"（主进程只做点包含判定，要便宜），
   但矩形可以按行合并出**多个**，精度远高于单一外接框。

2. **命中区必须包含 UI**
   它只有宠物本体。我们有工具栏、气泡、菜单、面板。
   → 上报的命中区 = 「她本体的行合并矩形」∪「所有 `.dshp-*` 可见元素的屏幕矩形」。
   菜单一开，命中区立刻变大，主进程下一次轮询（120ms）就把它变成可点。**死区问题就此消失。**

3. **拖动时暂停穿透刷新**
   拖动中窗口在动，命中区坐标会过期 → 会误判穿透、把拖动打断。
   拖动期间**固定为可点**（`setIgnoreMouseEvents(false)`），松手后再恢复轮询。

### 落地顺序（每步都能单独验证）

1. **先修聚焦**：窗口 `focusable:false`。独立、零风险、立刻见效。
2. **装回穿透（新写法）**：`refreshPetMousePassthrough` + 渲染进程上报命中矩形 + 值去重 + 1200ms guard。
   验收：左上角空白**点得到桌面**（死区消失），且**指针靠近她不再闪**。
3. **拖动换增量法**：`pet:moveBy(dx,dy)` / `pet:endDrag(localBounds)`，替掉反推循环。
   验收：上下左右都能跟手，y 轴不再冻住。
4. **删掉成型**：`window-shape.*`、`SetWindowRgn` 接线、`settings.json` 的 `shapeWindow`。
   验收：16 组测试改完后仍全绿（成型那组替换成穿透+拖动的守门员）。
5. **合并光标源 + 位置持久化按本体坐标**。

### 需要新增/改动的接口（草案）

```
渲染进程 → 主进程
  pet:updateHitBounds  (rects[])        // 屏幕坐标；她本体 + 所有可见 UI
  pet:moveBy           (dx, dy, localBounds, projectedBodyCenter)
  pet:endDrag          (localBounds)
  pet:guardInteraction (ms)             // 气泡/提醒弹出时调用
主进程 → 渲染进程
  pet:cursorMoved      ({x, y})         // 复用穿透那次轮询，驱动视线
删掉
  /dsh-pet/shape（掩码 POST）、window-shape.cjs/.mjs、main.js 里的成型接线
```

**⚠️ 不能丢的两条老教训**（换了机制也必须转移过去）：
- 拖动节拍若仍用 `setInterval` 会被 Windows 15.6ms 粒度拉长 → 增量法由 `pointermove` 驱动，
  天然没有这个问题，但**不能退回到"每帧读全局光标"**；
- `settings.json` 的 BOM 兜底、`clampPet` 按本体夹取，两条都保留。

## 三、还没验证的假设（动手前要先确认）

1. **`forward:true` 在本机到底转发什么**：NewMax 依赖它让主进程的视线/穿透在穿透状态下仍工作。
   我们交接文档里记的是"只转发移动、不转发按键"——**这与 NewMax 的用法不矛盾**
   （它也只靠它拿移动），但**按下**必须靠"切换成就可点"来保证。
   → 动手第 2 步时，先用真机日志确认：穿透状态下移动是否到达页面。
2. **120ms 轮询够不够快**：指针快速掠过她身体时，可能有一瞬（≤120ms）判成穿透、点击落空。
   NewMax 接受这个代价。我们要不要降到 60ms？先按 120ms 做，实测手感再调。
3. **命中区上报频率**：她一直在动，掩码每 1.5s 重采。上台频率若低于窗口移动频率，
   拖动中会误判。→ 所以第 3 条"拖动时暂停刷新"是必需的，不是优化。
